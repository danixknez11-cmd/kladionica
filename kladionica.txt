          import urllib.request, json, re, datetime

          api_key = "89138ea1afc9c5f5fef2f87cadb10d96"
          lige = ["soccer_uefa_champs_league", "soccer_epl", "soccer_spain_la_liga", "soccer_italy_serie_a"]
          out = []
          count = 0
          now = datetime.datetime.now(datetime.timezone.utc)

          for sport in lige:
              try:
                  url_sc = f"https://the-odds-api.com{sport}/scores/?apiKey={api_key}&daysFrom=2"
                  req = urllib.request.urlopen(url_sc, timeout=15)
                  scores = json.loads(req.read().decode("utf-8"))
                  for g in scores:
                      if g.get("completed") and g.get("scores"):
                          h, a = 0, 0
                          for s in g["scores"]:
                              if s["name"] == g["home_team"]: h = int(s["score"])
                              if s["name"] == g["away_team"]: a = int(s["score"])
                          ishod = 1 if h > a else (3 if a > h else 2)
                          out.append(f"REZ|{g[\"id\"]}|{ishod}")
              except Exception:
                  pass

              if count < 15:
                  try:
                      url_od = f"https://the-odds-api.com{sport}/odds/?apiKey={api_key}&regions=eu&markets=h2h"
                      req = urllib.request.urlopen(url_od, timeout=15)
                      matches = json.loads(req.read().decode("utf-8"))
                      for m in matches:
                          if count >= 15: break
                          if not m.get("bookmakers"): continue
                          ct = datetime.datetime.fromisoformat(m["commence_time"].replace("Z", "+00:00"))
                          if ct <= now: continue

                          outcomes = m["bookmakers"][0]["markets"][0]["outcomes"]
                          k1, kx, k2 = "1.50", "3.00", "1.50"
                          for o in outcomes:
                              if o["name"] == m["home_team"]: k1 = f"{float(o[\"price\"]):.2f}"
                              elif o["name"] == m["away_team"]: k2 = f"{float(o[\"price\"]):.2f}"
                              else: kx = f"{float(o[\"price\"]):.2f}"

                          home = re.sub(r"[^A-Za-z0-9 ]", "", m["home_team"])[:22]
                          away = re.sub(r"[^A-Za-z0-9 ]", "", m["away_team"])[:22]
                          out.append(f"MEC|{m[\"id\"]}|{home}|{away}|{k1}|{kx}|{k2}")
                          count += 1
                  except Exception:
                      pass

          if not out:
              out.append("NEMA_MECEVA_IL_ISTEKAO_KLJUC")

          with open("kladionica.txt", "w", encoding="utf-8") as f:
              f.write("\n".join(out) + "\n")
